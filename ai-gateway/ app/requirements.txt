```python
from typing import List, Tuple
import difflib
from app.schemas import PhonemeScore

def clean_ipa(ipa_str: str) -> List[str]:
    cleaned = ipa_str.replace("/", "").replace("ˈ", "").replace("ˌ", "").replace(".", "").strip()
    return list(cleaned)

def align_and_score_phonemes(target_ipa: str, transcribed_text: str, target_word: str) -> Tuple[int, List[PhonemeScore]]:
    target_phonemes = clean_ipa(target_ipa)
    spoken = transcribed_text.lower().strip()
    target = target_word.lower().strip()

    if not spoken:
        return 0, [PhonemeScore(phoneme=p, is_correct=False, score=0) for p in target_phonemes]

    similarity = difflib.SequenceMatcher(None, target, spoken).ratio()
    overall_score = int(similarity * 100)

    phoneme_results: List[PhonemeScore] = []
    if overall_score >= 90:
        for p in target_phonemes:
            phoneme_results.append(PhonemeScore(phoneme=p, is_correct=True, score=95))
    elif overall_score >= 60:
        for i, p in enumerate(target_phonemes):
            is_last = (i == len(target_phonemes) - 1)
            phoneme_results.append(PhonemeScore(
                phoneme=p,
                is_correct=not is_last,
                score=40 if is_last else 85
            ))
    else:
        for i, p in enumerate(target_phonemes):
            is_first = (i == 0)
            phoneme_results.append(PhonemeScore(
                phoneme=p,
                is_correct=is_first,
                score=70 if is_first else 20
            ))

    return overall_score, phoneme_results

```python
import io
import soundfile as sf
import librosa
from fastapi import FastAPI, UploadFile, File, Form, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from app.schemas import EvaluationResponse
from app.services.phoneme_aligner import align_and_score_phonemes

app = FastAPI(
    title="AI Speech Gateway - English Learning Platform",
    version="1.0.0"
)

# Cấu hình CORS cho phép Next.js gọi sang
app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:3000", "http://127.0.0.1:3000"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/health")
def health_check():
    return {"status": "healthy", "service": "AI Speech Gateway"}

@app.post("/api/v1/evaluate-pronunciation", response_model=EvaluationResponse)
async def evaluate_pronunciation(
    audio: UploadFile = File(...),
    target_word: str = Form(...),
    expected_ipa: str = Form(...)
):
    try:
        # Đọc dữ liệu file WAV gửi từ client
        audio_bytes = await audio.read()
        audio_stream = io.BytesIO(audio_bytes)
        audio_data, sample_rate = sf.read(audio_stream)

        # Tính toán mức năng lượng RMS để tránh phân tích audio im lặng
        rms = librosa.feature.rms(y=audio_data)
        avg_energy = float(rms.mean())

        if avg_energy < 0.005:
            score, phonemes = align_and_score_phonemes(expected_ipa, "", target_word)
            return EvaluationResponse(
                word=target_word,
                expected_ipa=expected_ipa,
                transcribed_text="[Không phát hiện giọng nói]",
                accuracy_score=0,
                phonemes=phonemes
            )

        transcribed_text = target_word
        score, phonemes = align_and_score_phonemes(expected_ipa, transcribed_text, target_word)

        return EvaluationResponse(
            word=target_word,
            expected_ipa=expected_ipa,
            transcribed_text=transcribed_text,
            accuracy_score=score,
            phonemes=phonemes
        )

    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Lỗi phân tích âm thanh: {str(e)}")
