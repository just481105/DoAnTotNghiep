=================== backend/src/auth/auth.service.ts ===================
import {
  BadRequestException,
  ConflictException,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { Prisma } from '@prisma/client';
import { compare, hash } from 'bcryptjs';
import { PrismaService } from '../prisma/prisma.service';
import { LoginDto } from './dto/login.dto';
import { RegisterDto } from './dto/register.dto';

const BCRYPT_ROUNDS = 12;

@Injectable()
export class AuthService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly jwtService: JwtService,
  ) {}

  async register(dto: RegisterDto) {
    if (Buffer.byteLength(dto.password, 'utf8') > 72) {
      throw new BadRequestException('Mật khẩu không được dài quá 72 byte.');
    }

    const email = dto.email.trim().toLowerCase();
    const passwordHash = await hash(dto.password, BCRYPT_ROUNDS);

    try {
      const user = await this.prisma.user.create({
        data: {
          email,
          passwordHash,
          fullName: dto.fullName?.trim() || null,
        },
        select: {
          id: true,
          email: true,
          fullName: true,
          role: true,
        },
      });

      return this.createSession(user);
    } catch (error) {
      if (
        error instanceof Prisma.PrismaClientKnownRequestError &&
        error.code === 'P2002'
      ) {
        throw new ConflictException('Email này đã được đăng ký.');
      }

      throw error;
    }
  }

  async login(dto: LoginDto) {
    if (Buffer.byteLength(dto.password, 'utf8') > 72) {
      throw new UnauthorizedException('Email hoặc mật khẩu không chính xác.');
    }

    const user = await this.prisma.user.findUnique({
      where: { email: dto.email.trim().toLowerCase() },
      select: {
        id: true,
        email: true,
        passwordHash: true,
        fullName: true,
        role: true,
      },
    });

    if (!user || !(await compare(dto.password, user.passwordHash))) {
      throw new UnauthorizedException('Email hoặc mật khẩu không chính xác.');
    }

    return this.createSession({
      id: user.id,
      email: user.email,
      fullName: user.fullName,
      role: user.role,
    });
  }

  private async createSession(user: {
    id: string;
    email: string;
    fullName: string | null;
    role: string;
  }) {
    const accessToken = await this.jwtService.signAsync({
      sub: user.id,
      email: user.email,
      role: user.role,
    });

    return {
      access_token: accessToken,
      user,
    };
  }
}

=================== backend/src/auth/guards/jwt-auth.guard.ts ===================
import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { Request } from 'express';

export interface AuthenticatedUser {
  sub: string;
  email: string;
  role: string;
}

type AuthenticatedRequest = Request & { user?: AuthenticatedUser };

function isAuthenticatedUser(value: unknown): value is AuthenticatedUser {
  if (typeof value !== 'object' || value === null) return false;

  return (
    'sub' in value &&
    typeof value.sub === 'string' &&
    value.sub.length > 0 &&
    'email' in value &&
    typeof value.email === 'string' &&
    'role' in value &&
    typeof value.role === 'string'
  );
}

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(private readonly jwtService: JwtService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest<AuthenticatedRequest>();
    const authorization = request.headers.authorization;
    const match =
      typeof authorization === 'string'
        ? /^Bearer\s+(\S+)$/i.exec(authorization)
        : null;

    if (!match) {
      throw new UnauthorizedException('A valid bearer token is required.');
    }

    let payload: unknown;
    try {
      payload = await this.jwtService.verifyAsync<Record<string, unknown>>(
        match[1],
      );
    } catch {
      throw new UnauthorizedException('The access token is invalid or expired.');
    }

    if (!isAuthenticatedUser(payload)) {
      throw new UnauthorizedException('The access token has an invalid payload.');
    }

    request.user = payload;
    return true;
  }
}

import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class ExamService {
  constructor(private readonly prisma: PrismaService) {}

  // 1. Lấy danh sách đề thi (kèm fallback nếu database chưa có)
  async getAllExams() {
    try {
      const exams = await (this.prisma as any).exam.findMany({
        where: { isPublished: true },
        include: {
          _count: { select: { questions: true } },
        },
        orderBy: { createdAt: 'desc' },
      });

      if (exams && exams.length > 0) {
        return exams;
      }
    } catch (err) {
      console.warn('Lỗi truy vấn danh sách đề thi, chuyển fallback:', err);
    }

    return [
      {
        id: 'exam-vstep-b2-01',
        title: 'VSTEP Reading Mock Test 01 (B1 - B2)',
        description: 'Đề thi thử kỹ năng Đọc hiểu định dạng VSTEP kiểm tra từ vựng ngữ cảnh kinh tế và xã hội.',
        examType: 'VSTEP',
        durationMin: 15,
        totalMarks: 30,
        _count: { questions: 3 },
      },
    ];
  }

  // 2. Lấy chi tiết đề thi và các câu hỏi trắc nghiệm
  async getExamById(id: string) {
    const exam = await (this.prisma as any).exam.findUnique({
      where: { id },
      include: {
        questions: {
          orderBy: { orderIndex: 'asc' },
          include: {
            options: {
              select: {
                id: true,
                optionLabel: true,
                content: true,
              },
            },
          },
        },
      },
    });

    if (!exam) throw new NotFoundException('Không tìm thấy đề thi');
    return exam;
  }

  // 3. Chấm điểm tự động và LƯU TRỰC TIẾP vào bảng user_exam_attempts
  async submitExam(userId: string, examId: string, userAnswers: Record<string, string>) {
    console.log(`\n========================================`);
    console.log(`📩 Nhận bài nộp: examId=${examId} | userId=${userId}`);

    const exam = await (this.prisma as any).exam.findUnique({
      where: { id: examId },
      include: {
        questions: {
          include: { options: true },
        },
      },
    });

    if (!exam) throw new NotFoundException('Không tìm thấy đề thi');

    let totalCorrect = 0;
    const reviewDetails = [];

    for (const q of exam.questions) {
      const selectedOptionId = userAnswers[q.id];
      const correctOption = q.options.find((opt: any) => opt.isCorrect);
      const isCorrect = selectedOptionId === correctOption?.id;

      if (isCorrect) totalCorrect++;

      reviewDetails.push({
        questionId: q.id,
        prompt: q.prompt,
        explanation: q.explanation,
        selectedOptionId: selectedOptionId || null,
        correctOptionId: correctOption?.id,
        isCorrect,
      });
    }

    const score = Math.round((totalCorrect / exam.questions.length) * 100);

    // Ghi dữ liệu vào PostgreSQL
    const attempt = await (this.prisma as any).userExamAttempt.create({
      data: {
        userId,
        examId,
        score,
        totalCorrect,
        completedAt: new Date(),
      },
    });

    console.log(`✅ ĐÃ LƯU KẾT QUẢ THI VÀO POSTGRESQL!`);
    console.log(`🆔 Mã lần thi: ${attempt.id} | Điểm: ${score}% | Số câu đúng: ${totalCorrect}/${exam.questions.length}`);
    console.log(`========================================\n`);

    return {
      attemptId: attempt.id,
      score,
      totalCorrect,
      totalQuestions: exam.questions.length,
      reviewDetails,
    };
  }
}

=================== backend/src/flashcards/dto/utils/sm2.utils.ts ===================
export interface SM2Input {
  quality: number; // 0 đến 5
  repetition: number;
  interval: number;
  easinessFactor: number;
}

export interface SM2Output {
  repetition: number;
  interval: number;
  easinessFactor: number;
  nextReviewDate: Date;
}

export function calculateSM2({
  quality,
  repetition,
  interval,
  easinessFactor,
}: SM2Input): SM2Output {
  let nextRepetition = repetition;
  let nextInterval = interval;
  let nextEF = easinessFactor;

  // Tính lại Easiness Factor: EF' = EF + (0.1 - (5 - q) * (0.08 + (5 - q) * 0.02))
  nextEF = nextEF + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02));
  if (nextEF < 1.3) nextEF = 1.3; // Ngưỡng sàn theo chuẩn SM-2

  if (quality < 3) {
    // Trả lời sai/quên: reset chu kỳ ôn về đầu
    nextRepetition = 0;
    nextInterval = 1;
  } else {
    // Trả lời đúng
    if (nextRepetition === 0) {
      nextInterval = 1;
    } else if (nextRepetition === 1) {
      nextInterval = 6;
    } else {
      nextInterval = Math.round(nextInterval * nextEF);
    }
    nextRepetition += 1;
  }

  const nextReviewDate = new Date();
  nextReviewDate.setDate(nextReviewDate.getDate() + nextInterval);

  return {
    repetition: nextRepetition,
    interval: nextInterval,
    easinessFactor: Number(nextEF.toFixed(2)),
    nextReviewDate,
  };
}

import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service'; // Chỉnh lại đường dẫn tới PrismaService của bạn nếu khác
import { calculateSM2 } from './dto/utils/sm2.utils';


@Injectable()
export class FlashcardsService {
  constructor(private prisma: PrismaService) {}

  async reviewCard(
    userId: string,
    wordId: string,
    rating: 'again' | 'hard' | 'good' | 'easy',
  ) {
    const qualityMap: Record<'again' | 'hard' | 'good' | 'easy', number> = {
      again: 1,
      hard: 3,
      good: 4,
      easy: 5,
    };
    const quality = qualityMap[rating];

    const current = await this.prisma.userFlashcard.upsert({
      where: { userId_wordId: { userId, wordId } },
      update: {},
      create: { userId, wordId },
    });

    const result = calculateSM2({
      quality,
      repetition: current.repetition,
      interval: current.interval,
      easinessFactor: current.easinessFactor,
    });

    return this.prisma.userFlashcard.update({
      where: { id: current.id },
      data: {
        repetition: result.repetition,
        interval: result.interval,
        easinessFactor: result.easinessFactor,
        nextReviewDate: result.nextReviewDate,
        lastReviewedAt: new Date(),
      },
    });
  }
}

=================== backend/src/grammar/grammar.service.ts ===================
import { Injectable, NotFoundException } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { FilterQuery, Model } from 'mongoose';
import { GrammarLesson, GrammarLessonDocument } from './grammar.schema';

const escapeRegex = (s: string) => s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

@Injectable()
export class GrammarService {
  constructor(
    @InjectModel(GrammarLesson.name)
    private readonly model: Model<GrammarLessonDocument>,
  ) {}

  private toDto(l: Record<string, any>, withContent = false) {
    const dto = {
      id: String(l._id),
      slug: l.slug,
      title: l.title,
      order: l.order,
      grade: l.grade ?? null,
      topic: l.topic ?? null,
      summary: l.summary ?? null,
      updatedAt: l.updatedAt,
    };
    return withContent ? { ...dto, content: l.content } : dto;
  }

  async findAll(filters: { grade?: unknown; topic?: unknown; q?: unknown }) {
    const where: FilterQuery<GrammarLessonDocument> = { isPublished: true };

    if (typeof filters.grade === 'string') {
      const grade = Number(filters.grade);
      if (Number.isInteger(grade)) where.grade = grade;
    }
    if (typeof filters.topic === 'string' && filters.topic) {
      where.topic = filters.topic;
    }
    if (typeof filters.q === 'string' && filters.q.trim()) {
      where.title = { $regex: escapeRegex(filters.q.trim()), $options: 'i' };
    }

    const rows = await this.model
      .find(where)
      .select('-content')
      .sort({ order: 1, createdAt: 1 })
      .lean();

    return rows.map((row) => this.toDto(row));
  }

  async findBySlug(slug: string) {
    const lesson = await this.model.findOne({ slug, isPublished: true }).lean();
    if (!lesson) throw new NotFoundException('Không tìm thấy bài học');
    return this.toDto(lesson, true);
  }
}

import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { ReviewGradeDto } from './dto/create-sm2.dto';

@Injectable()
export class Sm2Service {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * Hàm toán học tính toán các chỉ số SM-2
   */
  calculateSm2(
    quality: number,
    currentRepetitions: number,
    currentInterval: number,
    currentEf: number,
  ) {
    let repetitions = currentRepetitions;
    let intervalDays = currentInterval;
    let easinessFactor = currentEf;

    if (quality >= 3) {
      // Nhớ thành công
      if (repetitions === 0) {
        intervalDays = 1;
      } else if (repetitions === 1) {
        intervalDays = 6;
      } else {
        intervalDays = Math.round(intervalDays * easinessFactor);
      }
      repetitions += 1;
    } else {
      // Quên từ hoặc nhớ sai nghiêm trọng
      repetitions = 0;
      intervalDays = 1;
    }

    // Công thức cập nhật hệ số dễ EF chuẩn SuperMemo-2
    easinessFactor =
      easinessFactor + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02));

    // Ngưỡng chặn dưới của EF là 1.3
    if (easinessFactor < 1.3) {
      easinessFactor = 1.3;
    }

    // Làm tròn 2 chữ số thập phân
    easinessFactor = Math.round(easinessFactor * 100) / 100;

    // Tính mốc thời gian ôn tập kế tiếp (hiện tại + interval ngày)
    const nextReviewAt = new Date();
    nextReviewAt.setDate(nextReviewAt.getDate() + intervalDays);

    return {
      repetitions,
      intervalDays,
      easinessFactor,
      nextReviewAt,
    };
  }

  /**
   * API: Lấy danh sách từ vựng cần ôn tập hôm nay của học viên
   */
  async getDueWords(userId: string) {
    const now = new Date();

    return this.prisma.userSm2Stats.findMany({
      where: {
        userId,
        nextReviewAt: {
          lte: now, // Những từ có lịch ôn tập nhỏ hơn hoặc bằng hiện tại
        },
      },
      include: {
        vocabulary: true,
      },
      take: 20, // Giới hạn 20 thẻ mỗi phiên học
    });
  }

  /**
   * API: Ghi nhận điểm đánh giá (0-5) và cập nhật tiến trình SM-2 vào PostgreSQL
   */

async getAllVocabularies() {
  return this.prisma.vocabulary.findMany({
    orderBy: { createdAt: 'asc' },
  });
}


  async processReview(userId: string, dto: ReviewGradeDto) {
    const { vocabularyId, quality } = dto;

    // Tìm trạng thái học của từ vựng này trong DB
    let stat = await this.prisma.userSm2Stats.findUnique({
      where: {
        userId_vocabularyId: { userId, vocabularyId },
      },
    });

    // Nếu học lần đầu, khởi tạo các chỉ số mặc định
    const currentRep = stat ? stat.repetitions : 0;
    const currentInterval = stat ? stat.intervalDays : 0;
    const currentEf = stat ? stat.easinessFactor : 2.5;

    // Áp dụng thuật toán SM-2
    const calculated = this.calculateSm2(
      quality,
      currentRep,
      currentInterval,
      currentEf,
    );

    // Cập nhật hoặc thêm mới vào bảng user_sm2_stats
    return this.prisma.userSm2Stats.upsert({
      where: {
        userId_vocabularyId: { userId, vocabularyId },
      },
      update: {
        repetitions: calculated.repetitions,
        intervalDays: calculated.intervalDays,
        easinessFactor: calculated.easinessFactor,
        nextReviewAt: calculated.nextReviewAt,
        lastQuality: quality,
      },
      create: {
        userId,
        vocabularyId,
        repetitions: calculated.repetitions,
        intervalDays: calculated.intervalDays,
        easinessFactor: calculated.easinessFactor,
        nextReviewAt: calculated.nextReviewAt,
        lastQuality: quality,
      },
      include: {
        vocabulary: true,
      },
    });
  }

async getVocabularyTopics(userId: string) {
  // Lấy toàn bộ từ vựng kèm trạng thái SM-2 của học viên
  const vocabularies = await this.prisma.vocabulary.findMany({
    include: {
      sm2Stats: { where: { userId } },
    },
  });

  // Gom nhóm theo topic (tag)
  const topicMap: Record<string, { totalWords: number; learnedWords: number }> = {};

  for (const v of vocabularies) {
    const topic = v.topic || "Tổng quát";
    if (!topicMap[topic]) {
      topicMap[topic] = { totalWords: 0, learnedWords: 0 };
    }
    topicMap[topic].totalWords += 1;
    if (v.sm2Stats && v.sm2Stats.length > 0 && v.sm2Stats[0].repetitions > 0) {
      topicMap[topic].learnedWords += 1;
    }
  }

  // Trả về danh sách card giống ảnh mẫu
  return Object.keys(topicMap).map((topicName, index) => {
    const info = topicMap[topicName];
    const progress = info.totalWords > 0 ? Math.round((info.learnedWords / info.totalWords) * 100) : 0;
    return {
      id: `topic-${index + 1}`,
      slug: encodeURIComponent(topicName),
      title: topicName,
      desc: "Bắt đầu bài tập để ghi nhớ từ vựng mới hoặc ôn tập lại.",
      wordCount: info.totalWords,
      learnCount: 200 + index * 125, // Mock số lượt học cộng đồng
      progressPercent: progress,
      isLearned: progress > 0,
    };
  });
}

async getWordsByTopic(userId: string, topic?: string) {
  const whereCondition: any = {};
  if (topic && topic !== 'all') {
    whereCondition.topic = { equals: topic, mode: 'insensitive' };
  }

  const vocabularies = await this.prisma.vocabulary.findMany({
    where: whereCondition,
    include: {
      sm2Stats: {
        where: { userId },
      },
    },
    orderBy: { createdAt: 'asc' },
  });

  // Định dạng lại dữ liệu trả về cho Flashcard Engine
  return vocabularies.map((v) => {
    const stats = v.sm2Stats[0] || null;
    return {
      id: v.id,
      word: v.word,
      pos: v.pos,
      cefrLevel: v.cefrLevel,
      ipa: v.ipa,
      meaningVi: v.meaningVi,
      meaningEn: v.meaningEn,
      exampleEn: v.exampleEn,
      exampleVi: v.exampleVi,
      audioUrl: v.audioUrl,
      topic: v.topic,
      repetitions: stats ? stats.repetitions : 0,
      intervalDays: stats ? stats.intervalDays : 0,
      easinessFactor: stats ? stats.easinessFactor : 2.5,
    };
  });
}

        streakDays: true,
        xpPoints: true,

    // Từ đến hạn ôn tập hôm nay

      where: {
        userId,
        nextReviewAt: { lte: now },
      },
    });

    // Từ mới đang học (repetitions = 0)
    const learningCount = await this.prisma.userSm2Stats.count({
      where: {

        repetitions: 0,

    const activeLearners = await this.prisma.user.count();

import difflib
from app.schemas import PhonemeScore

def clean_ipa(ipa_str: str) -> List[str]:
    cleaned = ipa_str.replace("/", "").replace("ˈ", "").replace("ˌ", "").replace(".", "").strip()
    return list(cleaned)

def align_and_score_phonemes(target_ipa: str, transcribed_text: str, target_word: str) -> Tuple[int, List[PhonemeScore]]:
    target_phonemes = clean_ipa(target_ipa)
    spoken = transcribed_text.lower().strip()
    target = target_word.lower().strip()

    if not spoken:
        return 0, [PhonemeScore(phoneme=p, is_correct=False, score=0) for p in target_phonemes]

    similarity = difflib.SequenceMatcher(None, target, spoken).ratio()
    overall_score = int(similarity * 100)

    phoneme_results: List[PhonemeScore] = []
    if overall_score >= 90:
        for p in target_phonemes:
            phoneme_results.append(PhonemeScore(phoneme=p, is_correct=True, score=95))
    elif overall_score >= 60:
        for i, p in enumerate(target_phonemes):
            is_last = (i == len(target_phonemes) - 1)
            phoneme_results.append(PhonemeScore(
                phoneme=p,
                is_correct=not is_last,
                score=40 if is_last else 85
            ))
    else:
        for i, p in enumerate(target_phonemes):
            is_first = (i == 0)
            phoneme_results.append(PhonemeScore(
                phoneme=p,
                is_correct=is_first,
                score=70 if is_first else 20
            ))

    return overall_score, phoneme_results

@app.post("/api/v1/evaluate-pronunciation", response_model=EvaluationResponse)
async def evaluate_pronunciation(
    audio: UploadFile = File(...),
    target_word: str = Form(...),
    expected_ipa: str = Form(...)
):
    try:
        # Đọc dữ liệu file WAV gửi từ client
        audio_bytes = await audio.read()
        audio_stream = io.BytesIO(audio_bytes)
        audio_data, sample_rate = sf.read(audio_stream)

        # Tính toán mức năng lượng RMS để tránh phân tích audio im lặng
        rms = librosa.feature.rms(y=audio_data)
        avg_energy = float(rms.mean())

        if avg_energy < 0.005:
            score, phonemes = align_and_score_phonemes(expected_ipa, "", target_word)
            return EvaluationResponse(
                word=target_word,
                expected_ipa=expected_ipa,
                transcribed_text="[Không phát hiện giọng nói]",
                accuracy_score=0,
                phonemes=phonemes
            )

        transcribed_text = target_word
        score, phonemes = align_and_score_phonemes(expected_ipa, transcribed_text, target_word)

        return EvaluationResponse(
            word=target_word,
            expected_ipa=expected_ipa,
            transcribed_text=transcribed_text,
            accuracy_score=score,
            phonemes=phonemes
        )

    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Lỗi phân tích âm thanh: {str(e)}")

from typing import List, Tuple
import difflib
from app.schemas import PhonemeScore

def clean_ipa(ipa_str: str) -> List[str]:
    cleaned = ipa_str.replace("/", "").replace("ˈ", "").replace("ˌ", "").replace(".", "").strip()
    return list(cleaned)

def align_and_score_phonemes(target_ipa: str, transcribed_text: str, target_word: str) -> Tuple[int, List[PhonemeScore]]:
    target_phonemes = clean_ipa(target_ipa)
    spoken = transcribed_text.lower().strip()
    target = target_word.lower().strip()

    if not spoken:
        return 0, [PhonemeScore(phoneme=p, is_correct=False, score=0) for p in target_phonemes]

    similarity = difflib.SequenceMatcher(None, target, spoken).ratio()
    overall_score = int(similarity * 100)

    phoneme_results: List[PhonemeScore] = []
    if overall_score >= 90:
        for p in target_phonemes:
            phoneme_results.append(PhonemeScore(phoneme=p, is_correct=True, score=95))
    elif overall_score >= 60:
        for i, p in enumerate(target_phonemes):
            is_last = (i == len(target_phonemes) - 1)
            phoneme_results.append(PhonemeScore(
                phoneme=p,
                is_correct=not is_last,
                score=40 if is_last else 85
            ))
    else:
        for i, p in enumerate(target_phonemes):
            is_first = (i == 0)
            phoneme_results.append(PhonemeScore(
                phoneme=p,
                is_correct=is_first,
                score=70 if is_first else 20
            ))

    return overall_score, phoneme_results

const CORE_API_URL = (

  .replace(/\/api\/v1\/?$/, '')

export async function fetchWithAuth<T = unknown>(
  endpoint: string,
  options: RequestInit = {},
): Promise<T> {
  const token = typeof window !== 'undefined' ? localStorage.getItem('access_token') : null;

  const headers = {
    'Content-Type': 'application/json',
    ...(token ? { Authorization: `Bearer ${token}` } : {}),
    ...options.headers,
  };

  const response = await fetch(`${CORE_API_URL}${endpoint}`, {
    ...options,
    headers,
  });

  if (response.status === 401 && typeof window !== 'undefined') {
    localStorage.removeItem('access_token');
    window.dispatchEvent(new Event('auth:expired'));
  }

  if (!response.ok) {
    const errorData = await response.json().catch(() => ({}));
    throw new Error(errorData.message || 'Lỗi khi gọi API Core Engine');
  }

  return response.json() as Promise<T>;
}

function heatLevel(seed: number): number {
  const r = ((seed * 1103515245 + 12345) % 2147483648) / 2147483648;
  if (r > 0.96) return 3;
  if (r > 0.92) return 2;
  if (r > 0.86) return 1;
  return 0;
}

"use client";

import React, { useState, useEffect, useCallback, Suspense } from "react";
import { useSearchParams } from "next/navigation";
import Link from "next/link";
import { ArrowLeft, RefreshCw, Layers } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { fetchWithAuth } from "@/services/api-client";

interface VocabCard {
  id: string;
  word: string;
  pos: string;
  cefrLevel: string;
  ipa: string;
  meaningVi: string;
  meaningEn?: string;
  exampleEn: string;
  exampleVi?: string;
  audioUrl?: string;
  topic?: string;
  repetitions: number;
  intervalDays: number;
  easinessFactor: number;
}

function LearnContent() {
  const searchParams = useSearchParams();
  const currentTopic = searchParams.get("topic");

  const [cards, setCards] = useState<VocabCard[]>([]);
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isFlipped, setIsFlipped] = useState(false);
  const [isLoading, setIsLoading] = useState(true);
  const [isGrading, setIsGrading] = useState(false);

  // 1. Tải danh sách từ vựng theo Topic từ Backend
  const loadCards = useCallback(async () => {
    setIsLoading(true);
    try {
      const topicQuery = currentTopic
        ? `?topic=${encodeURIComponent(currentTopic)}`
        : "";
      const data = await fetchWithAuth<VocabCard[]>(
        `/sm2/words${topicQuery}`,
      );
      setCards(data);
      setCurrentIndex(0);
      setIsFlipped(false);
    } catch (err) {
      console.error(err);
    } finally {
      setIsLoading(false);
    }
  }, [currentTopic]);

  useEffect(() => {
    loadCards();
  }, [loadCards]);

  // 2. Phát âm thanh chuẩn US
  const handlePlayAudio = (word: string) => {
    if (typeof window !== "undefined" && "speechSynthesis" in window) {
      window.speechSynthesis.cancel();
      const utterance = new SpeechSynthesisUtterance(word);
      utterance.lang = "en-US";
      utterance.rate = 0.9;
      window.speechSynthesis.speak(utterance);
    }
  };

  // 3. Gửi đánh giá SM-2 (0 - 5 điểm)
  const handleGrade = async (grade: number) => {
    if (cards.length === 0 || isGrading) return;
    setIsGrading(true);

    const currentCard = cards[currentIndex];
    try {
      await fetchWithAuth("/sm2/grade", {
        method: "POST",
        body: JSON.stringify({
          vocabularyId: currentCard.id,
          quality: grade,
        }),
      });

      // Lật lại mặt trước và chuyển sang từ kế tiếp
      setIsFlipped(false);
      setTimeout(() => {
        if (currentIndex + 1 < cards.length) {
          setCurrentIndex((prev) => prev + 1);
        } else {
          setCurrentIndex(cards.length); // Hoàn thành bộ thẻ
        }
        setIsGrading(false);
      }, 200);
    } catch (err) {
      console.error("Lỗi khi lưu điểm SM-2:", err);
      setIsGrading(false);
    }
  };

  // Phím tắt bàn phím: Space để lật, phím 0-5 để chấm điểm
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.code === "Space") {
        e.preventDefault();
        setIsFlipped((prev) => !prev);
      } else if (["0", "1", "2", "3", "4", "5"].includes(e.key) && isFlipped) {
        handleGrade(parseInt(e.key, 10));
      }
    };
    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [isFlipped, currentIndex, cards]);

  if (isLoading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-slate-50 dark:bg-slate-950">
        <p className="text-sm font-semibold text-slate-500 animate-pulse">Đang nạp bộ Flashcard...</p>
      </div>
    );
  }

  const currentCard = cards[currentIndex];

  return (
    <div className="min-h-screen bg-slate-50/50 dark:bg-slate-950 flex flex-col justify-between p-4 sm:p-6 select-none">
      {/* HEADER: Quay lại + Thông tin chủ đề + Tiến độ */}
      <div className="max-w-xl w-full mx-auto flex items-center justify-between">
        <Link
          href="/vocab-sets"
          className="w-10 h-10 rounded-2xl border border-slate-200 dark:border-slate-800 flex items-center justify-center text-slate-600 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors"
        >
          <ArrowLeft className="w-5 h-5" />
        </Link>

        <div className="text-center">
          <Badge variant="outline" className="text-xs uppercase font-bold text-indigo-600 border-indigo-200">
            {currentTopic ? decodeURIComponent(currentTopic) : "Tất cả từ vựng"}
          </Badge>
          <p className="text-xs text-muted-foreground mt-0.5">
            {cards.length > 0 && currentIndex < cards.length ? `${currentIndex + 1} / ${cards.length}` : "Hoàn thành"}
          </p>
        </div>

        <button
          onClick={loadCards}
          className="w-10 h-10 rounded-2xl border border-slate-200 dark:border-slate-800 flex items-center justify-center text-slate-600 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors"
        >
          <RefreshCw className="w-4 h-4" />
        </button>
      </div>

      {/* THẺ FLASHCARD 3D */}
      <div className="max-w-xl w-full mx-auto my-auto py-6">
        {currentIndex < cards.length && currentCard ? (
          <div
            onClick={() => setIsFlipped(!isFlipped)}
            className="w-full min-h-[360px] bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-3xl p-8 shadow-sm flex flex-col justify-between cursor-pointer hover:shadow-md transition-all relative overflow-hidden"
          >
            {!isFlipped ? (
              /* MẶT TRƯỚC */
              <div className="flex flex-col justify-between h-full space-y-6">
                <div className="flex items-center justify-between">
                  <Badge className="bg-indigo-50 text-indigo-600 border-none font-bold">
                    {currentCard.cefrLevel || "B1"}
                  </Badge>
                  <span className="text-[11px] font-mono text-slate-400 uppercase tracking-widest">
                    {currentCard.pos}
                  </span>
                </div>

                <div className="text-center space-y-2">
                  <h2 className="text-3xl font-black text-slate-900 dark:text-slate-100 tracking-tight">
                    {currentCard.word}
                  </h2>
                  <p className="text-sm font-mono text-slate-500">{currentCard.ipa}</p>
                </div>

                <div className="flex justify-center">
                  <Button
                    size="sm"
                    variant="secondary"
                    className="rounded-xl text-xs gap-1.5"
                    onClick={(e) => {
                      e.stopPropagation();
                      handlePlayAudio(currentCard.word);
                    }}
                  >
                    🔊 Nghe phát âm
                  </Button>
                </div>

                <p className="text-[11px] text-center text-slate-400 italic">Nhấn Space hoặc chạm vào thẻ để lật</p>
              </div>
            ) : (
              /* MẶT SAU */
              <div className="flex flex-col justify-between h-full space-y-4">
                <div className="border-b border-slate-100 dark:border-slate-800 pb-3">
                  <h3 className="text-lg font-bold text-indigo-600 dark:text-indigo-400">
                    {currentCard.meaningVi}
                  </h3>
                  {currentCard.meaningEn && (
                    <p className="text-xs text-slate-500 mt-1">{currentCard.meaningEn}</p>
                  )}
                </div>

                <div className="space-y-1.5">
                  <p className="text-xs italic text-slate-700 dark:text-slate-300">
                    "{currentCard.exampleEn}"
                  </p>
                  {currentCard.exampleVi && (
                    <p className="text-[11px] text-slate-400">{currentCard.exampleVi}</p>
                  )}
                </div>

                <p className="text-[11px] text-center text-slate-400 italic">Chọn mức độ ghi nhớ bên dưới</p>
              </div>
            )}

function writeString(view: DataView, offset: number, string: string) {
  for (let i = 0; i < string.length; i++) {
    view.setUint8(offset + i, string.charCodeAt(i));
  }
}

function encodeWAV(samples: Float32Array, sampleRate: number = 16000): Blob {
  const buffer = new ArrayBuffer(44 + samples.length * 2);
  const view = new DataView(buffer);

  writeString(view, 0, "RIFF");
  view.setUint32(4, 36 + samples.length * 2, true);
  writeString(view, 8, "WAVE");
  writeString(view, 12, "fmt ");
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true); // Format = 1 (PCM)
  view.setUint16(22, 1, true); // Mono
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, sampleRate * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true); // 16-bit
  writeString(view, 36, "data");
  view.setUint32(40, samples.length * 2, true);

  let offset = 44;
  for (let i = 0; i < samples.length; i++, offset += 2) {
    const s = Math.max(-1, Math.min(1, samples[i]));
    view.setInt16(offset, s < 0 ? s * 0x8000 : s * 0x7fff, true);
  }

  return new Blob([view], { type: "audio/wav" });
}

  const queryWord = searchParams.get("word");

  const [targetList, setTargetList] = useState<TargetWordItem[]>(DEFAULT_TARGET_WORDS);
  const [wordIndex, setWordIndex] = useState(0);
  const [isRecording, setIsRecording] = useState(false);
  const [isEvaluating, setIsEvaluating] = useState(false);
  const [evalResult, setEvalResult] = useState<SpeechEvaluationResult | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [xpEarned, setXpEarned] = useState<number | null>(null);

  const analyserRef = useRef<AnalyserNode | null>(null);
  const processorRef = useRef<ScriptProcessorNode | null>(null);
  const streamRef = useRef<MediaStream | null>(null);
  const recordedSamplesRef = useRef<Float32Array[]>([]);
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const animationFrameRef = useRef<number | null>(null);

  const initWordFromQuery = useCallback(async () => {
    if (!queryWord) return;

    try {
      const foundIdx = targetList.findIndex(
        (item) => item.word.toLowerCase() === queryWord.toLowerCase()
      );

      if (foundIdx !== -1) {
        setWordIndex(foundIdx);
        return;
      }

      const words = await fetchWithAuth<VocabularyLookup[]>("/sm2/words");
      if (Array.isArray(words)) {
        const matched = words.find(
          (w) => w.word.toLowerCase() === queryWord.toLowerCase()
        );

        if (matched) {
          const newTarget: TargetWordItem = {
            id: matched.id,
            word: matched.word,
            pos: matched.pos || "word",
            cefr: matched.cefrLevel || "B2",
            ipa: matched.ipa || "/.../",
            meaning: matched.meaningVi || matched.meaningEn || "",
          };
          setTargetList([newTarget, ...DEFAULT_TARGET_WORDS]);
          setWordIndex(0);
        }
      }
    } catch (err) {
      console.warn("Không thể đồng bộ từ mục tiêu từ URL:", err);
    }
  }, [queryWord, targetList]);

  useEffect(() => {
    initWordFromQuery();
  }, [initWordFromQuery]);

  const currentTarget = targetList[wordIndex] || DEFAULT_TARGET_WORDS[0];

  const drawWaveform = () => {
    if (!analyserRef.current || !canvasRef.current) return;
    const canvas = canvasRef.current;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const bufferLength = analyserRef.current.frequencyBinCount;
    const dataArray = new Uint8Array(bufferLength);

    const render = () => {
      animationFrameRef.current = requestAnimationFrame(render);
      analyserRef.current!.getByteTimeDomainData(dataArray);

      ctx.fillStyle = "rgb(15 23 42)";
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      ctx.lineWidth = 2;
      ctx.strokeStyle = "rgb(56 189 248)";
      ctx.beginPath();

      const sliceWidth = (canvas.width * 1.0) / bufferLength;
      let x = 0;

      for (let i = 0; i < bufferLength; i++) {
        const v = dataArray[i] / 128.0;
        const y = (v * canvas.height) / 2;

        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);

        x += sliceWidth;
      }

      ctx.lineTo(canvas.width, canvas.height / 2);
      ctx.stroke();
    };

    render();
  };

  const startRecording = async () => {
    try {
      setError(null);
      setEvalResult(null);
      setXpEarned(null);
      recordedSamplesRef.current = [];

      const stream = await navigator.mediaDevices.getUserMedia({
        audio: {
          channelCount: 1,
          sampleRate: 16000,
          echoCancellation: true,
          noiseSuppression: true,
        },
      });
      streamRef.current = stream;

      const AudioContextClass =
        window.AudioContext || (window as any).webkitAudioContext;
      const audioCtx = new AudioContextClass({ sampleRate: 16000 });
      audioContextRef.current = audioCtx;

      const source = audioCtx.createMediaStreamSource(stream);
      const analyser = audioCtx.createAnalyser();
      analyser.fftSize = 256;
      analyserRef.current = analyser;

      const processor = audioCtx.createScriptProcessor(4096, 1, 1);
      processorRef.current = processor;

      processor.onaudioprocess = (e) => {
        const inputData = e.inputBuffer.getChannelData(0);
        recordedSamplesRef.current.push(new Float32Array(inputData));
      };

      source.connect(analyser);
      analyser.connect(processor);
      processor.connect(audioCtx.destination);

      drawWaveform();
      setIsRecording(true);
    } catch (err: any) {
      console.error(err);
      setError("Không thể truy cập Micro. Hãy cấp quyền trên trình duyệt.");
    }
  };

    if (!isRecording) return;
    setIsRecording(false);

    if (animationFrameRef.current) cancelAnimationFrame(animationFrameRef.current);
    if (streamRef.current) streamRef.current.getTracks().forEach((t) => t.stop());
    if (processorRef.current) processorRef.current.disconnect();
    if (audioContextRef.current && audioContextRef.current.state !== "closed") {
      audioContextRef.current.close();
    }

    const totalLength = recordedSamplesRef.current.reduce((acc, c) => acc + c.length, 0);
    const merged = new Float32Array(totalLength);
    let offset = 0;
    for (const chunk of recordedSamplesRef.current) {
      merged.set(chunk, offset);
      offset += chunk.length;
    }

    const wavBlob = encodeWAV(merged, 16000);
    await sendAudioToFastApi(wavBlob);
  };

        method: "POST",
        body: formData,
      });

      if (!response.ok) {
        const errDetail = await response.text();
        throw new Error(`Mã lỗi ${response.status}: ${errDetail}`);
      }

      try {

            accuracyScore: data.accuracy_score,
            transcribedText: data.transcribed_text,
          }),

      setIsEvaluating(false);
    }

    setError(null);

    <main className="max-w-2xl mx-auto px-4 py-10 flex flex-col items-center">

            {currentTarget.pos} • {currentTarget.cefr}

          <span className="text-xs text-muted-foreground">

        <div className="flex items-center justify-center gap-3">
          <h1 className="text-4xl font-extrabold tracking-tight text-slate-900 dark:text-white">
            {currentTarget.word}
          </h1>

            type="button"

                    }`}

                  onClick={() => setPage(currentPage + 1)}
                  className="grid size-9 place-items-center rounded-xl border border-slate-200 bg-white text-slate-600 transition hover:bg-slate-50 disabled:opacity-40"

import React, { useCallback, useEffect, useState } from "react";
import Link from "next/link";

const BACKEND_URL = process.env.NEXT_PUBLIC_CORE_API_URL || "http://localhost:5001";

interface QuestionOption {

  optionLabel: string;
  content: string;

  id: string;
  prompt: string;
  orderIndex: number;
  options: QuestionOption[];

interface ExamDetail {
  id: string;
  title: string;
  description: string;
  durationMin: number;
  questions: Question[];
}

interface ReviewItem {
  questionId: string;
  prompt: string;
  explanation: string;
  selectedOptionId: string | null;
  correctOptionId: string;
  isCorrect: boolean;
}

interface ExamResult {
  score: number;
  totalCorrect: number;
  totalQuestions: number;
  reviewDetails: ReviewItem[];
}

export default function ExamTakingPage() {
  const params = useParams();
  const examId = params?.id as string;

  const [exam, setExam] = useState<ExamDetail | null>(null);
  const [selectedAnswers, setSelectedAnswers] = useState<Record<string, string>>({});
  const [timeLeft, setTimeLeft] = useState<number>(0);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [result, setResult] = useState<ExamResult | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  // 1. Tải chi tiết đề thi từ NestJS
  useEffect(() => {
    if (!examId) return;

    async function fetchExam() {
      try {
        const res = await fetch(`${BACKEND_URL}/exams/${examId}`);
        if (res.ok) {
          const data = await res.json();
          setExam(data);
          setTimeLeft((data.durationMin || 15) * 60);
        }
      } catch (err) {
        console.error("Lỗi nạp đề thi:", err);
      } finally {
        setIsLoading(false);
      }
    }
    fetchExam();
  }, [examId]);

  // 3. Gửi bài làm sang NestJS để chấm điểm và lưu vào CSDL
  const handleSubmitExam = useCallback(async () => {
    if (isSubmitting || result || !examId) return;
    setIsSubmitting(true);
    try {
      const data = await fetchWithAuth<ExamResult>(`/exams/${examId}/submit`, {
        method: "POST",
        body: JSON.stringify({
          answers: selectedAnswers,
        }),
      });

      setResult(data);
    } catch (err) {
      console.error("Lỗi gửi bài thi:", err);
    } finally {
      setIsSubmitting(false);
    }

  useEffect(() => {
    if (timeLeft <= 0 || result) return;
    const timer = setInterval(() => {
      setTimeLeft((prev) => {
        if (prev <= 1) {
          clearInterval(timer);
          void handleSubmitExam();
          return 0;
        }
        return prev - 1;
      });
    }, 1000);
    return () => clearInterval(timer);
  }, [timeLeft, result, handleSubmitExam]);

  const formatTime = (seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins.toString().padStart(2, "0")}:${secs.toString().padStart(2, "0")}`;

                        key={opt.id}

            <div className="p-3.5 rounded-2xl bg-rose-50 text-rose-600 dark:bg-rose-950/40 dark:text-rose-400 text-xs flex items-center gap-2">
              <AlertTriangle className="w-4 h-4 shrink-0" />
              <span>{errorMsg}</span>
            </div>
          )}

          <div className="flex items-center justify-end gap-3 pt-2">
            <Button
              variant="outline"
              size="sm"
              onClick={handleReset}
              disabled={isAnalyzing || !essayContent}
              className="rounded-xl text-xs gap-1.5"
            >
              <RotateCcw className="w-3.5 h-3.5" /> Viết lại
            </Button>
            <Button
              size="sm"
              onClick={handleEvaluate}
              disabled={isAnalyzing || wordCount < 20}
              className="rounded-xl text-xs font-bold gap-1.5 bg-indigo-600 hover:bg-indigo-700 text-white"
            >
              {isAnalyzing ? (
                <>
                  <Loader2 className="w-3.5 h-3.5 animate-spin" /> AI đang chấm điểm...
                </>
              ) : (
                <>
                  <Sparkles className="w-3.5 h-3.5" /> Chấm điểm với AI
                </>
              )}
            </Button>
          </div>
        </div>

        {/* Khung Kết Quả Đánh Giá Chi Tiết */}
        {result && (
          <div className="space-y-6 animate-in fade-in-50 duration-500">
            {/* Điểm tổng quan và 4 tiêu chí */}
            <div className="bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-3xl p-6 sm:p-8 shadow-sm space-y-6">
              <div className="flex flex-col sm:flex-row items-center justify-between pb-6 border-b border-slate-100 dark:border-slate-800 gap-4">
                <div>
                  <h3 className="text-xl font-bold text-slate-900 dark:text-slate-100">
                    Báo Cáo Đánh Giá Bài Viết
                  </h3>
                  <p className="text-xs text-slate-400 mt-0.5">
                    Thang điểm 10 quy đổi chuẩn định dạng VSTEP
                  </p>
                </div>
                <div className="flex items-center gap-3">
                  <span className="text-xs font-semibold text-slate-400 uppercase">
                    Điểm Tổng:
                  </span>
                  <span className="text-4xl font-black text-indigo-600 dark:text-indigo-400">
                    {result.overall_score}
                    <span className="text-base font-normal text-slate-400">/10</span>
                  </span>
                </div>
              </div>

              {/* 4 Tiêu chí VSTEP */}
              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
                <div className="p-4 rounded-2xl bg-slate-50 dark:bg-slate-800/60 border text-center space-y-1">
                  <span className="text-[11px] font-bold text-slate-400 block">Bố cục & Nhiệm vụ</span>
                  <span className="text-xl font-extrabold text-slate-800 dark:text-slate-200">
                    {result.task_achievement}
                  </span>
                </div>
                <div className="p-4 rounded-2xl bg-slate-50 dark:bg-slate-800/60 border text-center space-y-1">
                  <span className="text-[11px] font-bold text-slate-400 block">Độ mạch lạc</span>
                  <span className="text-xl font-extrabold text-slate-800 dark:text-slate-200">
                    {result.coherence_cohesion}
                  </span>
                </div>
                <div className="p-4 rounded-2xl bg-slate-50 dark:bg-slate-800/60 border text-center space-y-1">
                  <span className="text-[11px] font-bold text-slate-400 block">Vốn từ vựng</span>
                  <span className="text-xl font-extrabold text-slate-800 dark:text-slate-200">
                    {result.lexical_resource}
                  </span>
                </div>
                <div className="p-4 rounded-2xl bg-slate-50 dark:bg-slate-800/60 border text-center space-y-1">
                  <span className="text-[11px] font-bold text-slate-400 block">Độ chuẩn ngữ pháp</span>
                  <span className="text-xl font-extrabold text-slate-800 dark:text-slate-200">
                    {result.grammatical_accuracy}
                  </span>
                </div>
              </div>

              {/* Nhận xét tổng quan */}
              <div className="p-4 rounded-2xl bg-indigo-50/50 dark:bg-indigo-950/30 border border-indigo-100 dark:border-indigo-900 text-xs text-indigo-900 dark:text-indigo-300 leading-relaxed flex items-start gap-3">
                <Lightbulb className="w-5 h-5 text-indigo-600 shrink-0 mt-0.5" />
                <div>
                  <strong className="font-bold">Nhận xét của AI Mentor:</strong> {result.general_feedback}
                </div>
              </div>
            </div>

            {/* Chi tiết sửa lỗi ngữ pháp */}
            <div className="bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-3xl p-6 shadow-sm space-y-4">
              <h4 className="text-sm font-bold text-slate-900 dark:text-slate-100 flex items-center gap-2">
                <CheckCircle2 className="w-4 h-4 text-emerald-500" /> Các lỗi ngữ pháp cần lưu ý
              </h4>

              {result.grammar_errors.length === 0 ? (
                <p className="text-xs text-slate-500 italic">Không phát hiện lỗi ngữ pháp cơ bản nào.</p>
              ) : (
                <div className="space-y-3">
                  {result.grammar_errors.map((err, i) => (
                    <div key={i} className="p-4 rounded-2xl border border-slate-100 dark:border-slate-800 bg-slate-50/50 dark:bg-slate-950 space-y-1.5 text-xs">
                      <div className="flex items-center gap-2 flex-wrap">
                        <span className="text-rose-600 line-through bg-rose-50 dark:bg-rose-950/60 px-2 py-0.5 rounded-lg font-mono font-medium">
                          {err.original_text}
                        </span>
                        <span className="text-slate-400">→</span>
                        <span className="text-emerald-600 font-bold bg-emerald-50 dark:bg-emerald-950/60 px-2 py-0.5 rounded-lg font-mono">
                          {err.corrected_text}
                        </span>
                      </div>
                      <p className="text-slate-500 dark:text-slate-400">{err.explanation}</p>
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* Đoạn văn tham khảo viết lại tự nhiên hơn */}
            <div className="bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-3xl p-6 shadow-sm space-y-3">
              <h4 className="text-sm font-bold text-slate-900 dark:text-slate-100 flex items-center gap-2">
                <Sparkles className="w-4 h-4 text-amber-500" /> Phiên bản viết lại tự nhiên hơn (Native Rewrite)
              </h4>
              <div className="p-4 rounded-2xl bg-amber-50/40 dark:bg-amber-950/20 border border-amber-200/50 dark:border-amber-900/50 text-xs text-slate-700 dark:text-slate-300 leading-relaxed font-sans italic">
                "{result.better_version}"
              </div>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
